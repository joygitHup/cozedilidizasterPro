```python
    call_detail = models.JSONField(default=dict, blank=True)
```

### 4.3 预警服务 - 核心预警引擎 (warning_service/apps/engines/rule_engine.py)

```python
import logging
from typing import Dict, Any
from django.utils import timezone
from django.db import transaction
from .models import WarningRecord
from .ml_predictor import MLPredictor
from monitor_service.client import MonitorClient

logger = logging.getLogger(__name__)

class WarningEngine:
    """预警引擎 - 核心业务逻辑"""
    
    def __init__(self):
        self.monitor_client = MonitorClient()
        self.ml_predictor = MLPredictor()
    
    def evaluate(self, hazard_point_id: str) -> Dict[str, Any]:
        """
        评估该隐患点是否触发预警
        返回: {triggered: bool, level: str, confidence: float, reason: str}
        """
        # 1. 获取实时监测数据
        latest_data = self.monitor_client.get_latest_data(hazard_point_id)
        if not latest_data:
            return {'triggered': False, 'reason': '无监测数据'}
        
        # 2. 规则引擎判断（快速阈值）
        rule_result = self._check_thresholds(latest_data)
        
        # 3. 机器学习模型判断（深度分析）
        ml_result = self.ml_predictor.predict(hazard_point_id, latest_data)
        
        # 4. 多源融合决策
        result = self._fuse_results(rule_result, ml_result)
        
        # 5. 如果触发预警，保存记录
        if result['triggered']:
            self._create_warning_record(hazard_point_id, result)
            # 触发叫应
            self._trigger_call(result)
        
        return result
    
    def _check_thresholds(self, data: Dict) -> Dict:
        """阈值规则检查"""
        alerts = []
        # 牛顿力突降检查
        if data.get('npr_force') and data['npr_force']['change_rate'] > 30:
            alerts.append({'type': 'npr_drop', 'level': 'red'})
        # 降雨量检查
        if data.get('rainfall', 0) > 50:  # 50mm/小时
            alerts.append({'type': 'rainfall', 'level': 'orange'})
        # 位移速率检查
        if data.get('displacement_rate', 0) > 10:  # 10mm/天
            alerts.append({'type': 'displacement', 'level': 'yellow'})
        
        return {
            'triggered': len(alerts) > 0,
            'level': max([a['level'] for a in alerts]) if alerts else None,
            'reasons': alerts
        }
    
    def _fuse_results(self, rule_result: Dict, ml_result: Dict) -> Dict:
        """多源结果融合"""
        # 规则触发 或 模型置信度>85% 则预警
        if rule_result['triggered'] or ml_result.get('confidence', 0) > 0.85:
            # 取最高等级
            levels = []
            if rule_result.get('level'):
                levels.append(rule_result['level'])
            if ml_result.get('level'):
                levels.append(ml_result['level'])
            
            # 等级优先级: red > orange > yellow > blue
            priority = {'red': 4, 'orange': 3, 'yellow': 2, 'blue': 1}
            final_level = max(levels, key=lambda x: priority.get(x, 0))
            
            return {
                'triggered': True,
                'level': final_level,
                'confidence': max(rule_result.get('confidence', 0.5), 
                                 ml_result.get('confidence', 0.5)),
                'reason': f"规则触发: {rule_result.get('reasons')} + 模型预测置信度{ml_result.get('confidence', 0)}"
            }
        return {'triggered': False}
    
    @transaction.atomic
    def _create_warning_record(self, hazard_point_id: str, result: Dict):
        """创建预警记录"""
        record = WarningRecord.objects.create(
            hazard_point_id=hazard_point_id,
            level=result['level'],
            trigger_type='auto',
            trigger_value=result,
            confidence=result['confidence'],
            status=WarningRecord.Status.PENDING
        )
        logger.info(f"预警触发: {record.id} - {record.hazard_point.name}")
        return record
    
    def _trigger_call(self, result: Dict):
        """触发叫应（通过Celery异步执行）"""
        from .tasks import send_alert_notification
        send_alert_notification.delay(result)
```

```python
    # 2. 缓存最新值到Redis（供预警服务快速读取）
    cache_key = f"latest:{data.device_id}:{data.data_type}"
    redis_client.setex(cache_key, 300, f"{data.value}|{data.timestamp.isoformat()}")
    
    # 3. 检查是否触发预警（可异步触发）
    # 触发预警引擎检查
    
    return {"status": "ok", "timestamp": data.timestamp}

@app.get("/api/monitor/data/{device_id}")
async def query_data(
    device_id: str, 
    data_type: str,
    start_time: str,
    end_time: str
):
    """查询监测数据"""
    # 从TDengine查询
    result = await query_tdengine(device_id, data_type, start_time, end_time)
    return {"data": result}
```

## 六、服务间通信

### 6.1 同步通信 - HTTP客户端封装

```python
# shared/common/service_client.py
import httpx
from typing import Optional
import logging

logger = logging.getLogger(__name__)

class ServiceClient:
    """微服务HTTP客户端"""
    
    SERVICE_MAP = {
        'auth': 'http://auth-service:8001',
        'hazard': 'http://hazard-service:8002',
        'monitor': 'http://monitor-service:8003',
        'warning': 'http://warning-service:8004',
        'emergency': 'http://emergency-service:8005',
        'govern': 'http://govern-service:8006',
        'ai': 'http://ai-service:8007',
    }
    
    def __init__(self, service_name: str):
        self.base_url = self.SERVICE_MAP.get(service_name)
        if not self.base_url:
            raise ValueError(f"Unknown service: {service_name}")
        self.client = httpx.Client(timeout=10.0)
    
    def get(self, path: str, params: Optional[dict] = None):
        """GET请求"""
        url = f"{self.base_url}{path}"
        logger.debug(f"调用服务: {url}")
        return self.client.get(url, params=params)
    
    def post(self, path: str, data: Optional[dict] = None):
        """POST请求"""
        url = f"{self.base_url}{path}"
        return self.client.post(url, json=data)
```

```python
        APPNotifier.push(contact['user_id'], warning_data)

    logger.info(f"预警通知已发送: {warning_data['id']}")
```
